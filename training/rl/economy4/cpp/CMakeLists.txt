cmake_minimum_required(VERSION 3.18)
project(zhanguo_cpp LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(zhanguo_core STATIC
    army.cpp bank.cpp deferred.cpp diplo.cpp economy.cpp mapgen.cpp market.cpp
    opponent_economy.cpp opponent_military.cpp opponents.cpp
    rl_candidates.cpp rl_env.cpp rollout.cpp save.cpp turn.cpp w3.cpp world.cpp)
target_include_directories(zhanguo_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(zhanguo main.cpp replay.cpp)
target_link_libraries(zhanguo PRIVATE zhanguo_core)

option(ZHANGUO_TORCH "Build the LibTorch sampler" OFF)
if(ZHANGUO_TORCH)
    find_package(OpenMP REQUIRED)
    find_package(Threads REQUIRED)
    set(ZHANGUO_TORCH_WHEEL "" CACHE PATH "Path to the torch package in a PyTorch wheel")
    set(ZHANGUO_TORCH_ABI "1" CACHE STRING "PyTorch wheel: 1 for the C++11 ABI and 0 for the old ABI")
    if(ZHANGUO_TORCH_WHEEL)
        set(TORCH_FLAGS "-D_GLIBCXX_USE_CXX11_ABI=${ZHANGUO_TORCH_ABI}")
        set(TORCH_LIBRARIES "")
        foreach(lib torch torch_cpu c10 torch_cuda c10_cuda)
            set(path "${ZHANGUO_TORCH_WHEEL}/lib/lib${lib}.so")
            if(EXISTS "${path}")
                list(APPEND TORCH_LIBRARIES "${path}")
            endif()
        endforeach()
    else()
        find_package(Torch REQUIRED)
        separate_arguments(TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
    endif()
    target_compile_options(zhanguo_core PUBLIC ${TORCH_FLAGS})
    add_executable(zhanguo_torch torch_sampler.cpp)
    add_executable(zhanguo_dual torch_dual_sampler.cpp)
    add_executable(zhanguo_script script_sampler.cpp)
    target_link_libraries(zhanguo_script PRIVATE zhanguo_core "${TORCH_LIBRARIES}" Threads::Threads)
    target_compile_features(zhanguo_dual PRIVATE cxx_std_20)
    target_link_libraries(zhanguo_dual PRIVATE zhanguo_core "${TORCH_LIBRARIES}" OpenMP::OpenMP_CXX Threads::Threads)
    target_compile_features(zhanguo_torch PRIVATE cxx_std_20)
    target_link_libraries(zhanguo_torch PRIVATE zhanguo_core "${TORCH_LIBRARIES}" OpenMP::OpenMP_CXX Threads::Threads)
    if(ZHANGUO_TORCH_WHEEL)
        target_include_directories(zhanguo_script SYSTEM PRIVATE "${ZHANGUO_TORCH_WHEEL}/include")
        target_include_directories(zhanguo_torch SYSTEM PRIVATE
            "${ZHANGUO_TORCH_WHEEL}/include" "${ZHANGUO_TORCH_WHEEL}/include/torch/csrc/api/include")
        target_link_options(zhanguo_torch PRIVATE -Wl,--no-as-needed)
        target_include_directories(zhanguo_dual SYSTEM PRIVATE
            "${ZHANGUO_TORCH_WHEEL}/include" "${ZHANGUO_TORCH_WHEEL}/include/torch/csrc/api/include")
        target_link_options(zhanguo_dual PRIVATE -Wl,--no-as-needed)
    endif()
endif()
